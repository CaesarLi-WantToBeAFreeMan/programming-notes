```java
    import java.time.LocalDate;
    import java.time.DayOfWeek;
    import java.time.format.DateTimeFormatter;
    import java.time.temporal.ChronoUnit;
    import java.time.temporal.TemporalAdjusters;

    public class Main{
        public static void main(String [] args){
            //create LocalDate instances
            LocalDate today = LocalDate.now();
            LocalDate christmas2024 = LocalDate.of(2024, 12, 25);
            LocalDate firstOf2025 = LocalDate.parse("2025-01-01");

            //getters
            System.out.printf(
                "today is:\t%s\nyear of today:\t%d\nmonth of today:\t%d\nday of month of today:\t%d\nday of week of today:\t%s\n",
                today,
                today.getYear(),
                today.getMonthValue(),
                today.getDayOfMonth(),
                today.getDayOfWeek()
            );

            //modifications
            System.out.printf(
                "next week:\t%s\nlast month:\t%s\nnext month:\t%s\nfirst day of the month:\t%s\nnext friday:\t%s\n",
                today.plusWeeks(1),
                today.minusMonths(1),
                today.plus(1, ChronoUnit.MONTHS),
                today.withDayOfMonth(1),
                today.with(TemporalAdjusters.next(DayOfWeek.FRIDAY))
            );

            //comparison
            System.out.printf(
                "Christmas in 2024 is before today:\t%b\nJanuary 1, 2025 is after today:\t%b\n",
                christmas2024.isBefore(today),
                firstOf2025.isAfter(today)
            );

            //convert
            System.out.printf(
                "start of today:\t%s\na moment of today:\t%s\nformatted string:\t%s\n",
                today.atStartOfDay(),
                today.atTime(12, 30),
                today.format(DateTimeFormatter.ofPattern("MM dd, yy"))
            );

            //other
            System.out.printf(
                "is this year a leap year:\t%b\nDays until 2026:\t%s\n",
                today.isLeapYear(),
                today.until(LocalDate.of(2026, 1, 1))
            );
        }
    }
```

```java
    import java.time.LocalTime;
    import java.time.temporal.ChronoField;
    import java.time.temporal.ChronoUnit;
    import java.time.format.DateTimeFormatter;

    public class Main{
        public static void main(String [] args){
            //create instances
            LocalTime now = LocalTime.now();
            System.out.println("now:\t" + now);
            LocalTime time0 = LocalTime.of(12, 45, 30, 750_123_456);
            System.out.printf(
                "hour of time0:\t%d\nminute of time0:\t%d\nsecond of time0:\t%d\nnanosecond of time:\t%d\n",
                time0.getHour(),
                time0.getMinute(),
                time0.getSecond(),
                time0.getNano()
            );
            LocalTime parsed = LocalTime.parse("06:04");
            System.out.println("parsed:\t" + parsed);

            //comparisons
            System.out.printf(
                "is time0 equals midnight:\t%b\nis time0 before now:\t%b\nis time0 after now:\t%b\n",
                time0.equals(now),
                time0.isBefore(now),
                time0.isAfter(now)
            );

            //modifications
            time0.plusHours(1);
            time0.plus(15, ChronoUnit.MINUTES);
            System.out.println("after adding 75 minutes:\t" + time0);
            time0.withNano(0);
            time0.with(ChronoField.SECOND_OF_MINUTE, 0);
            System.out.println("after setting nanosecond and second to 0:\t" + time0);

            //conversion
            System.out.printf(
                "nanosecond of day:\t%d\nformat string:\t%s\n",
                time0.toNanoOfDay(),
                time0.format(DateTimeFormatter.ofPattern("HH:mm:ss"))
            );

            //extra methods

            System.out.printf(
                "time0 is supported for ChronoFielf.SECOND_OF_MINUTE:\t%b\nrange of minute of hour:\t%s\ntruncate to minutes:\t%s\n",
                time0.isSupported(ChronoField.SECOND_OF_MINUTE),
                time0.range(ChronoField.MINUTE_OF_HOUR),
                time0.truncatedTo(ChronoUnit.MINUTES)
            );
        }
    }
```

            LocalDateTime time0 = LocalDateTime.of(2024, 12, 25, 10, 30);
            System.out.printf(
                "year of time0:\t%d\nmonth of time0:\t%s\nday of month of time0:\t%d\nday of week of time0:\t%s\nhour of time0:\t%d\nminute of time0:\t%d\nsecond of time0:\t%d\nnanosecond of time0:\t%d\n",

                time0.getMonth(),
                time0.getDayOfMonth(),
                time0.getDayOfWeek(),
                time0.getHour(),
                time0.getMinute(),
                time0.getSecond(),
                time0.getNano()
            );
            LocalDateTime parsed = LocalDateTime.parse("2025-01-01T12:00:00");
            System.out.println("parsed:\t" + parsed);

            //modifications
            System.out.println("adding 87,129 seconds:\t: " + now.plusDays(1).plusMinutes(12).plus(9, ChronoUnit.SECONDS));

            //comparisons
            System.out.printf(
                "is time0 after now:\t%b\nis time0 before now:\t%b\nis time0 equals now:\t%b\n",
                time0.isAfter(now),
                time0.isBefore(now),
                time0.isEqual(now)
            );

            //conversion to LocalDate and LocalTime
            System.out.printf(
                "date part of today:\t%s\ntime part of today:\t%s\n",
                now.toLocalDate(),
                now.toLocalTime()
            );

            //formatting
            System.out.println("formatted now:\t" + now.format(DateTimeFormatter.ofPattern("MM dd, yy HH:mm:ss")));
        }
    }
